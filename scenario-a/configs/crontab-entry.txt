# A3 task 11 -- install with `crontab -e` as root, or:
#   sudo cp healthcheck.sh /usr/local/bin/ && sudo chmod 755 /usr/local/bin/healthcheck.sh
#   sudo mkdir -p /etc/healthcheck && sudo cp checks.conf /etc/healthcheck/
#   sudo touch /var/log/healthcheck.log && sudo chmod 640 /var/log/healthcheck.log
#   sudo crontab -l 2>/dev/null | cat - crontab-entry.txt | sudo crontab -
#
# cron runs with a near-empty environment: PATH is normally just
# /usr/bin:/bin, so `flock`, `curl` and `awk` resolve but anything installed in
# /usr/local/bin does not. Setting PATH explicitly is why this works from cron
# and not just from my shell.
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
SHELL=/bin/bash

# stdout goes to /dev/null because the script already writes the detail to
# /var/log/healthcheck.log; stderr is kept so cron mails real breakage.
*/5 * * * * /usr/local/bin/healthcheck.sh /etc/healthcheck/checks.conf >/dev/null
